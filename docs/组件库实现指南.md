## 通用组件实现

### 1. CommonScaffold 组件

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

class CommonScaffold extends ConsumerWidget {
  final String title;
  final List<Widget> actions;
  final Widget body;
  final Widget? floatingActionButton;
  final AppBarSearchState? searchState;

  const CommonScaffold({
    super.key,
    required this.title,
    required this.actions,
    required this.body,
    this.floatingActionButton,
    this.searchState,
  });

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    return Scaffold(
      appBar: AppBar(
        title: Text(title),
        actions: actions,
        // 搜索功能集成
        bottom: searchState != null
            ? PreferredSize(
                preferredSize: const Size.fromHeight(56),
                child: Padding(
                  padding: const EdgeInsets.all(8.0),
                  child: TextField(
                    decoration: InputDecoration(
                      hintText: '搜索...',
                      prefixIcon: const Icon(Icons.search),
                      border: OutlineInputBorder(
                        borderRadius: BorderRadius.circular(8),
                      ),
                    ),
                    onChanged: searchState!.onSearch,
                  ),
                ),
              )
            : null,
      ),
      body: body,
      floatingActionButton: floatingActionButton,
    );
  }
}

class AppBarSearchState {
  final Function(String) onSearch;
  
  const AppBarSearchState({required this.onSearch});
}
```

### 2. 状态指示器组件

```dart
import 'package:flutter/material.dart';

enum ConnectionStatus { connecting, connected, disconnected }

class ConnectionStatusIndicator extends StatelessWidget {
  final ConnectionStatus status;
  final VoidCallback? onTap;
  final String connectingText;
  final String connectedText;
  final String disconnectedText;

  const ConnectionStatusIndicator({
    super.key,
    required this.status,
    this.onTap,
    this.connectingText = '连接中',
    this.connectedText = '已连接',
    this.disconnectedText = '未连接',
  });

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    
    Color backgroundColor;
    Color foregroundColor;
    Widget icon;
    String text;

    switch (status) {
      case ConnectionStatus.connecting:
        backgroundColor = theme.colorScheme.primaryContainer;
        foregroundColor = theme.colorScheme.onPrimaryContainer;
        icon = SizedBox(
          width: 20,
          height: 20,
          child: CircularProgressIndicator(
            strokeWidth: 2,
            color: foregroundColor,
          ),
        );
        text = connectingText;
      case ConnectionStatus.connected:
        backgroundColor = Colors.greenAccent;
        foregroundColor = theme.brightness == Brightness.light
            ? theme.colorScheme.onSurfaceVariant
            : theme.colorScheme.onPrimaryFixedVariant;
        icon = const Icon(Icons.check, weight: FontWeight.w900);
        text = connectedText;
      case ConnectionStatus.disconnected:
        backgroundColor = theme.colorScheme.error;
        foregroundColor = theme.colorScheme.onError;
        icon = const Icon(Icons.restart_alt_sharp, weight: FontWeight.w900);
        text = disconnectedText;
    }

    return FilledButton.icon(
      onPressed: onTap,
      style: FilledButton.styleFrom(
        visualDensity: VisualDensity.compact,
        padding: const EdgeInsets.symmetric(horizontal: 12),
        backgroundColor: backgroundColor,
        foregroundColor: foregroundColor,
      ),
      icon: SizedBox(
        width: 20,
        height: 20,
        child: icon,
      ),
      label: Text(text),
    );
  }
}
```

### 3. 代理列表项组件

```dart
import 'package:flutter/material.dart';

class ProxyListItem extends StatelessWidget {
  final String name;
  final int? delay;
  final bool isSelected;
  final VoidCallback onTap;
  final Widget? leading;
  final Widget? trailing;

  const ProxyListItem({
    super.key,
    required this.name,
    this.delay,
    required this.isSelected,
    required this.onTap,
    this.leading,
    this.trailing,
  });

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    
    return Card(
      elevation: isSelected ? 2 : 1,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(12),
        side: isSelected
            ? BorderSide(color: theme.colorScheme.primary, width: 2)
            : BorderSide.none,
      ),
      child: ListTile(
        leading: leading ?? _buildDefaultLeading(),
        title: Text(
          name,
          style: theme.textTheme.bodyMedium?.copyWith(
            fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,
          ),
        ),
        subtitle: Text(
          delay != null ? '${delay}ms' : '--',
          style: theme.textTheme.bodySmall?.copyWith(
            color: theme.colorScheme.onSurfaceVariant,
          ),
        ),
        trailing: trailing,
        onTap: onTap,
        contentPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
      ),
    );
  }

  Widget _buildDefaultLeading() {
    return Container(
      width: 40,
      height: 40,
      decoration: BoxDecoration(
        color: Colors.blue.shade100,
        shape: BoxShape.circle,
      ),
      child: Icon(
        Icons.cloud,
        color: Colors.blue.shade600,
        size: 20,
      ),
    );
  }
}
```

### 4. 网格布局组件

```dart
import 'package:flutter/material.dart';

class AdaptiveGrid extends StatelessWidget {
  final List<Widget> children;
  final double crossAxisSpacing;
  final double mainAxisSpacing;
  final EdgeInsets padding;

  const AdaptiveGrid({
    super.key,
    required this.children,
    this.crossAxisSpacing = 16,
    this.mainAxisSpacing = 16,
    this.padding = const EdgeInsets.all(16),
  });

  @override
  Widget build(BuildContext context) {
    final screenWidth = MediaQuery.of(context).size.width;
    
    // 自适应列数
    int crossAxisCount;
    if (screenWidth < 600) {
      crossAxisCount = 2; // 移动端
    } else if (screenWidth < 1200) {
      crossAxisCount = 4; // 平板端
    } else {
      crossAxisCount = 6; // 桌面端
    }

    return Padding(
      padding: padding,
      child: GridView.count(
        crossAxisCount: crossAxisCount,
        crossAxisSpacing: crossAxisSpacing,
        mainAxisSpacing: mainAxisSpacing,
        shrinkWrap: true,
        physics: const NeverScrollableScrollPhysics(),
        children: children,
      ),
    );
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

class DashboardPageTemplate extends ConsumerStatefulWidget {
  final String title;
  final List<Widget> dashboardWidgets;
  final List<Widget>? actions;
  final Widget? floatingActionButton;

  const DashboardPageTemplate({
    super.key,
    required this.title,
    required this.dashboardWidgets,
    this.actions,
    this.floatingActionButton,
  });

  @override
  ConsumerState<DashboardPageTemplate> createState() => _DashboardPageTemplateState();
}

class _DashboardPageTemplateState extends ConsumerState<DashboardPageTemplate> {
  bool _isEditMode = false;

  void _toggleEditMode() {
    setState(() {
      _isEditMode = !_isEditMode;
    });
  }

  @override
  Widget build(BuildContext context) {
    return CommonScaffold(
      title: widget.title,
      actions: [
        ...?widget.actions,
        IconButton(
          icon: Icon(_isEditMode ? Icons.save : Icons.edit),
          onPressed: _toggleEditMode,
        ),
      ],
      body: _isEditMode ? _buildEditMode() : _buildViewMode(),
      floatingActionButton: widget.floatingActionButton,
    );
  }

  Widget _buildViewMode() {
    return AdaptiveGrid(
      children: widget.dashboardWidgets,
    );
  }

  Widget _buildEditMode() {
    return Column(
      children: [
        Padding(
          padding: const EdgeInsets.all(16),
          child: Text(
            '拖拽调整组件位置',
            style: Theme.of(context).textTheme.bodyMedium?.copyWith(
              color: Theme.of(context).colorScheme.onSurfaceVariant,
            ),
          ),
        ),
        Expanded(
          child: AdaptiveGrid(
            children: widget.dashboardWidgets,
          ),
        ),
      ],
    );
  }
}
```

### 2. 列表页面模板

```dart
import 'package:flutter/material.dart';

class ListPageTemplate extends StatelessWidget {
  final String title;
  final List<Widget> items;
  final List<Widget>? actions;
  final Widget? floatingActionButton;
  final Function(String) onSearch;
  final Widget? emptyState;

  const ListPageTemplate({
    super.key,
    required this.title,
    required this.items,
    required this.onSearch,
    this.actions,
    this.floatingActionButton,
    this.emptyState,
  });

  @override
  Widget build(BuildContext context) {
    return CommonScaffold(
      title: title,
      actions: actions,
      searchState: AppBarSearchState(onSearch: onSearch),
      body: items.isEmpty
          ? emptyState ?? _buildDefaultEmptyState()
          : ListView.builder(
              padding: const EdgeInsets.all(16),
              itemCount: items.length,
              itemBuilder: (context, index) => Padding(
                padding: const EdgeInsets.only(bottom: 8),
                child: items[index],
              ),
            ),
      floatingActionButton: floatingActionButton,
    );
  }

  Widget _buildDefaultEmptyState() {
    return Center(
      child: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Icon(
            Icons.inbox,
            size: 64,
            color: Colors.grey.shade400,
          ),
          const SizedBox(height: 16),
          Text(
            '暂无数据',
            style: TextStyle(
              fontSize: 18,
              color: Colors.grey.shade600,
            ),
          ),
        ],
      ),
    );
  }
}
```

### 2. 响应式工具类

```dart
import 'package:flutter/material.dart';

class ResponsiveUtils {
  static bool isMobile(BuildContext context) {
    return MediaQuery.of(context).size.width < 600;
  }

  static bool isTablet(BuildContext context) {
    final width = MediaQuery.of(context).size.width;
    return width >= 600 && width < 1200;
  }

  static bool isDesktop(BuildContext context) {
    return MediaQuery.of(context).size.width >= 1200;
  }

  static double responsiveValue({
    required BuildContext context,
    required double mobile,
    required double tablet,
    required double desktop,
  }) {
    if (isMobile(context)) return mobile;
    if (isTablet(context)) return tablet;
    return desktop;
  }

  static int responsiveGridColumns(BuildContext context) {
    return responsiveValue(
      context: context,
      mobile: 2,
      tablet: 4,
      desktop: 6,
    );
  }

  static EdgeInsets responsivePadding(BuildContext context) {
    final base = responsiveValue(
      context: context,
      mobile: 16.0,
      tablet: 24.0,
      desktop: 32.0,
    );
    return EdgeInsets.all(base);
  }
}
```
